    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const canvas = document.getElementById('preview-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            
            const textInput = document.getElementById('text-input');
            const textColor = document.getElementById('text-color');
            const fontSizeInput = document.getElementById('font-size');
            const downloadBtn = document.getElementById('download-btn');
            const showCircle = document.getElementById('show-circle');
            const canvasWrapper = document.getElementById('canvas-wrapper');
            const watermarkOpacity = document.getElementById('watermark-opacity');
            const opacityValLabel = document.getElementById('opacity-val');
            const imageUpload = document.getElementById('image-upload');
            const uploadStatus = document.getElementById('upload-status');
            const userImg = document.getElementById('user-image');

            const currentBg = '#FFFFFF';
            let hasImage = false;

            function updateCanvas() {
                const size = 800; 
                const fullText = textInput.value || "預覽文字";
                const color = textColor.value;
                const fontSize = parseInt(fontSizeInput.value);
                const opacity = parseInt(watermarkOpacity.value) / 100;

                if (opacityValLabel) opacityValLabel.textContent = `${Math.round(opacity * 100)}%`;

                // 1. 白色背景
                ctx.fillStyle = currentBg;
                ctx.fillRect(0, 0, size, size);

                // 2. 繪製上傳的照片
                if (hasImage && userImg.complete && userImg.naturalWidth > 0) {
                    ctx.save();
                    ctx.globalAlpha = opacity;
                    
                    const targetSize = size * 0.82;
                    let dw, dh;
                    const ratio = userImg.width / userImg.height;

                    if (ratio > 1) {
                        dw = targetSize;
                        dh = targetSize / ratio;
                    } else {
                        dh = targetSize;
                        dw = targetSize * ratio;
                    }

                    ctx.drawImage(userImg, (size - dw) / 2, (size - dh) / 2, dw, dh);
                    ctx.restore();
                }

                // 3. 繪製文字
                ctx.save();
                ctx.shadowColor = 'rgba(0, 0, 0, 0.1)';
                ctx.shadowBlur = 10;
                ctx.fillStyle = color;
                ctx.font = `900 ${fontSize}px "Noto Sans TC", sans-serif`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                
                const maxWidth = size * 0.86;
                const lines = [];
                const rawLines = fullText.split('\n');

                rawLines.forEach(rawLine => {
                    let currentLine = '';
                    const chars = rawLine.split('');
                    for (let i = 0; i < chars.length; i++) {
                        let testLine = currentLine + chars[i];
                        if (ctx.measureText(testLine).width > maxWidth && i > 0) {
                            lines.push(currentLine);
                            currentLine = chars[i];
                        } else {
                            currentLine = testLine;
                        }
                    }
                    lines.push(currentLine);
                });

                const lineHeight = fontSize * 1.32;
                const totalH = lines.length * lineHeight;
                let startY = (size - totalH) / 2 + (lineHeight / 2);

                lines.forEach((line, index) => {
                    ctx.fillText(line, size / 2, startY + (index * lineHeight));
                });
                ctx.restore();
            }

            // 處理上傳
            if (imageUpload) {
                imageUpload.addEventListener('change', (e) => {
                    const file = e.target.files[0];
                    if (file) {
                        const reader = new FileReader();
                        reader.onload = (event) => {
                            userImg.onload = () => {
                                hasImage = true;
                                if (uploadStatus) {
                                    uploadStatus.textContent = "✅ 已成功載入照片檔案";
                                    uploadStatus.classList.add('text-green-600');
                                }
                                updateCanvas();
                            };
                            userImg.src = event.target.result;
                        };
                        reader.readAsDataURL(file);
                    }
                });
            }

            // 監聽變動
            [textInput, textColor, fontSizeInput, watermarkOpacity].forEach(el => {
                if (el) el.addEventListener('input', updateCanvas);
            });

            if (showCircle) {
                showCircle.addEventListener('change', (e) => {
                    if (canvasWrapper) canvasWrapper.classList.toggle('circle-mask', e.target.checked);
                });
            }

            // 下載
            if (downloadBtn) {
                downloadBtn.addEventListener('click', () => {
                    const link = document.createElement('a');
                    const fileName = textInput.value ? textInput.value.replace(/\n/g, '_') : 'LINE_Cover';
                    link.download = `${fileName}_${Date.now()}.png`;
                    link.href = canvas.toDataURL('image/png');
                    link.click();
                });
            }

            updateCanvas();
        });
    </script>
